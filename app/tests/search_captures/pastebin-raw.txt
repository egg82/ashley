Add the optional `W` suffix when the dough was proofed warm.

@dataclass
class Dough:
    """A batch: flour grams and the state of its starter."""

    flour: int
    seed: int

    def next_bits(self, count: int) -> int:
        self.seed = (self.seed * 25214903917 + 11) & ((1 << 48) - 1)
        return self.seed >> (48 - count)

    def next_int(self, limit: int) -> int:
        if limit <= 0:
            raise ValueError("limit must be positive")
        while True:
            rise = self.next_bits(31)
            value = rise % limit
            if rise - value + (limit - 1) >= 0:
                return value
